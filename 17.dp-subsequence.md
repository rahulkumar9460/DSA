# CASE 1. Longest Increasing Subsequence

State:
> dp[i] = Length of LIS ending at i
>
> Best subsequence whose LAST element is nums[i]

Transition:
> For every previous index:
>
>           for(int j=0; j<i; j++){
>               if(nums[j] < nums[i])
>                   dp[i] = max(dp[i], dp[j] + 1);
>           }

## 1. Longest Increasing Subsequence
[Leetcode link](https://leetcode.com/problems/longest-increasing-subsequence/description/)

Given an integer array nums, return the length of the longest strictly increasing subsequence.

 
Input: 
- nums = [10,9,2,5,3,7,101,18]
Output: 
- 4
Explanation: 
- The longest increasing subsequence is [2,3,7,101], therefore the length is 4.

```cpp
int lengthOfLIS(vector<int>& nums) {
    // dp[i] = LIS ending at i
    int n = nums.size();
    vector<int> dp(n, 1);

    int ans = 1;
    for(int i=1; i<n; i++) {
        for(int j=0; j<i; j++) {
            if(nums[j] < nums[i]) {
                dp[i] = max(dp[i], dp[j]+1);
            }
        }
        
        if(ans < dp[i]) ans = dp[i];
    }

    return ans;
}
/*
    Time = O(n^2)
*/
```

```cpp
int lengthOfLIS(vector<int>& nums) {
    vector<int> tails;
    int ans = 1;

    for(int num : nums) {
        if(tails.size() == 0 || tails.back() < num) tails.push_back(num);
        else {
            auto it = lower_bound(tails.begin(), tails.end(), num);
            *it = num;
        }

        if(ans < tails.size())ans = tails.size();
    }

    return ans;
}
```

## 2. Number of Longest Increasing Subsequence
[Leetcode link](https://leetcode.com/problems/number-of-longest-increasing-subsequence/description/)

Input: 
- nums = [1,3,5,4,7]
Output: 
- 2
Explanation: 
- The two longest increasing subsequences are [1, 3, 4, 7] and [1, 3, 5, 7].

Input: 
- nums = [2,2,2,2,2]
Output: 
- 5
Explanation: 
- The length of the longest increasing subsequence is 1, and there are 5 increasing subsequences of length 1, so output 5.

### Intuition
> [!IMPORTANT]
> For counting problems:
>
>           dp[i]    = longest length ending at i
>           cnt[i]   = number of ways to achieve dp[i]
>
> If better length found:
>
>           if dp[j] + 1 > dp[i]
>               dp[i] = dp[j] + 1;
>               cnt[i] = cnt[j];
>
> If same length found:
>
>           if dp[j] + 1 == dp[i]
>               cnt[i] += cnt[j]

```cpp
int findNumberOfLIS(vector<int>& nums) {
    int n = nums.size();
    vector<int> dp(n, 1);
    vector<int> cnt(n, 1);

    int maxLen = 1;

    for(int i=1; i<n; i++) {
        for(int j=0; j<i; j++) {
            if(nums[j] < nums[i]) {

                if(dp[i] == dp[j] + 1) // same length found
                    cnt[i] += cnt[j]; 

                else if(dp[i] < dp[j]+1){ // better length found
                    dp[i] = dp[j]+1;
                    cnt[i] = cnt[j];
                }

            }
        }

        if(maxLen < dp[i]) maxLen = dp[i];
    }

    int ans = 0;
    for(int i=0; i<n; i++)
        if(dp[i] == maxLen) 
            ans += cnt[i];
    
    return ans;
}
```

## 3. Largest Divisible Subset
[Leetcode link](https://leetcode.com/problems/largest-divisible-subset/description/)

Given a set of distinct positive integers nums, return the largest subset answer 
such that every pair (answer[i], answer[j]) of elements in this subset satisfies:

answer[i] % answer[j] == 0, or
answer[j] % answer[i] == 0
If there are multiple solutions, return any of them.

Input: 
- nums = [1,2,3]
Output: 
- [1,2]
Explanation: 
- [1,3] is also accepted.

### Intuition
> [!IMPORTANT]
> If we sort the nums now we for a pair we just need to check if
> - nums[i] % nums[j] == 0 , i > j
>               
>               We dont need to check for every pair
>               Since nums is sorted we just need to check for consecutive pairs
>               a, b, c, d ==> a < b < c < d
>               
>               b%a = 0, c%b = 0, d%c = 0
>               a divides b, b divides c ==> implies a divides c
>
>               Since we need to construct the subset store parent of each indicies
>
>                dp[i] = max number of largestDivisibleSubset ending at i

```cpp
vector<int> largestDivisibleSubset(vector<int>& nums) {
    
    sort(nums.begin(), nums.end());
    int n = nums.size();
    vector<int> dp(n, 0);
    vector<int> parent(n, -1);

    int maxLen = 0;
    int lastIdx = 0;

    for(int i=1; i<n; i++) {
        for(int j=0; j<i; j++) {
            if(nums[i]%nums[j] == 0) {
                
                if(dp[j] + 1 > dp[i]) {
                    dp[i] = dp[j] + 1;
                    parent[i] = j;
                }
            }
        }

        if(dp[i] > maxLen) {
            maxLen = dp[i];
            lastIdx = i;
        }
    }

    vector<int> ans;
    while(lastIdx >= 0) {
        ans.push_back(nums[lastIdx]);
        lastIdx = parent[lastIdx];
    }

    return ans;
}
```

## 4. Longest String Chain
[Leetcode link](https://leetcode.com/problems/longest-string-chain/description/)

word A is a predecessor of word B if and only if we can insert exactly one letter anywhere in word A 
without changing the order of the other characters to make it equal to word B.

 "abc" is a predecessor of "abac", while "cba" is not a predecessor of "bcad".

Return the length of the longest possible word chain with words chosen from the given list of words.

Input: 
- words = ["a","b","ba","bca","bda","bdca"]
Output: 
- 4
Explanation: 
- One of the longest word chains is ["a","ba","bda","bdca"].

### Intuition
> [!IMPORTANT]
> Sort the words with increasing size
>
> dp[i] = max length of chain ending at i
>
> transition:
>
>           for all prev j < i
>               if words[j] is predecessor of words[i]
>                   dp[i] = max(dp[i], dp[j]+1)

```cpp
bool comp(string &a, string &b) {
    return a.size() < b.size();
}

class Solution {
public:
    bool isPred(string &a, string &b) {
        int n = a.size(), m = b.size();
        if(n+1 != m) return false;

        int i = 0, j= 0;
        bool used = false;
        while(i<n && j< m) {
            if(a[i] != b[j]) {
                if(used) return false;

                used = true;
                j++;
            } else {
                i++;
                j++;
            }
        }

        return true;
    }
    
    int longestStrChain(vector<string>& words) {
        int n = words.size();
        sort(words.begin(), words.end(), comp);

        vector<int> dp(n, 1);
        int ans = 1;

        for(int i=1; i<n; i++) {
            for(int j=0; j<i; j++) {
                if(isPred(words[j], words[i])) {
                    if(dp[j] + 1 > dp[i])
                        dp[i] = dp[j] + 1;
                }
            }

            ans = max(ans, dp[i]);
        }

        return ans;
    }
};
```

## 5. Longest Arithmetic Subsequence
[Leetcode link](https://leetcode.com/problems/longest-arithmetic-subsequence/description/)

Given an array nums of integers, return the length of the longest arithmetic subsequence in nums.

Note that:

A subsequence is an array that can be derived from another array by deleting some or no elements
without changing the order of the remaining elements.
A sequence seq is arithmetic if seq[i + 1] - seq[i] are all the same value (for 0 <= i < seq.length - 1).

Input: 
- nums = [9,4,7,2,10]
Output: 
- 3
Explanation:  
- The longest arithmetic subsequence is [4,7,10].

### Intuition
> [!IMPORTANT]
> If we have dp[i]
>
>           max len of airth subsequence ending at i
>           Now we want to include k > i index
>                   Can we really decide
>                   We can't decide until we know the
>                   subsequence ending at i has what diff
>
> So now state becomes;
>
>           dp[i][diff]
>           max len of airth subsequence with difference diff
>
>           Diff can be negative: 
>               since nums[i] <= 500
>           
>           If diff < 0 ==> diff = -diff + 501

```cpp
int longestArithSeqLength(vector<int>& nums) {
    
    int n = nums.size();
    int max_ = 501;
    vector<vector<int>> dp(n, vector<int>(2*max_, 1));
    int ans = 2;

    for(int i=1; i<n; i++) {
        for(int j=0; j<i; j++) {
            int diff = nums[i] - nums[j];
            if(diff < 0) diff = -1*diff + max_;

            if(dp[j][diff] + 1 > dp[i][diff]) {
                dp[i][diff] = dp[j][diff] + 1;
            }

            ans = max(ans, dp[i][diff]);
        }
    }

    return ans;
}
```

## 6. Minimum Number of Removals to Make Mountain Array
[Leetcode link](https://leetcode.com/problems/minimum-number-of-removals-to-make-mountain-array/description/)

You may recall that an array arr is a mountain array if and only if:

arr.length >= 3

There exists some index i (0-indexed) with 0 < i < arr.length - 1 such that:
arr[0] < arr[1] < ... < arr[i - 1] < arr[i]
arr[i] > arr[i + 1] > ... > arr[arr.length - 1]
Given an integer array nums​​​, return the minimum number of elements to remove to make nums​​​ a mountain array.

- Input: nums = [2,1,1,5,6,2,3,1]
- Output: 3
- Explanation: One solution is to remove the elements at indices 0, 1, and 5, making the array nums = [1,5,6,3,1].

### Intuition
> [!IMPORTANT]
> Index i can be mountain peak if:
>
>               all elements to left are strictly increasing
>               all elements to right are strictly decreasing
>
>               incDp[i] = longest increasing subsequence ending at i
>               decDp[i] = longest decreasing subsequence starting at i
>
>               then length of mountain = incDp[i] + decDp[i] - 1
>
>               Elements removed = n - mountain len

```cpp
nt minimumMountainRemovals(vector<int>& nums) {
    int n = nums.size();
    vector<int> incDp(n, 1);
    vector<int> decDp(n, 1);

    for(int i=1; i<n; i++) {
        for(int j=0; j<i; j++) {
            if(nums[i] > nums[j])
                incDp[i] = max(incDp[i], incDp[j]+1);
        }
    }

    for(int i=n-2; i>=0; i--) {
        for(int j=n-1; j>i; j--) {
            if(nums[i] > nums[j])
                decDp[i] = max(decDp[i], decDp[j]+1);
        }
    }

    int ans = n;
    for(int i=1; i<n-1; i++) {
        if(incDp[i] > 1 && decDp[i] > 1) {

            int mountainLen = incDp[i] + decDp[i] - 1;
            ans = min(ans, n - mountainLen);
        }
    }

    return ans;
}
```

# CASE 2: Longest Common Subsequence

> [!NOTE]
> 
>               "I have two sequences, and I need to compare prefixes of both sequences."
>
>               Whenever you see that, your brain should immediately think:
>               2D DP on (i,j)

Given:
1. text1 = "abcde"
2. text2 = "ace"

> Find longest common subsequence.

> [!IMPORTANT]
>
> State:
>
>           dp[i][j]
>           LCS length between text1[0..i-1] and text2[0..j-1]
>
>           or:
>               first i chars of text1
>               first j chars of text2
>
> At every state i, j we have two possibilities:
>
>           1. Characters match, text1[i-1] == text2[j-1]
>               Example: abc, dec ===> c == c
>               Then we must take this character.
>                   dp[i][j] = 1 + dp[i-1][j-1];
>
>           2. Characters don't match, text1[i-1] != text2[j-1]
>              Example: abc, abd ===> c != d
>              Then One of them is useless. either skip c or skip d
>              Take the best:
>                   dp[i][j] = max(dp[i-1][j], dp[i][j-1]);
>

```cpp
int longestCommonSubsequence(string s1, string s2) {
    int n = s1.size();
    int m = s2.size();

    vector<vector<int>> dp(n+1, vector<int>(m+1, 0));

    for(int i=1; i<=n; i++) {
        for(int j=1; j<=m; j++) {

            if(s1[i-1] == s2[j-1])
                dp[i][j] = 1 + dp[i-1][j-1];
            else
                dp[i][j] =
                    max(dp[i-1][j],
                        dp[i][j-1]);
        }
    }

    return dp[n][m];
}
```

## 4. Minimum ASCII Delete Sum for Two Strings
[Leetcode link](https://leetcode.com/problems/minimum-ascii-delete-sum-for-two-strings/description/)

Given two strings s1 and s2, return the lowest ASCII sum of deleted characters to make two strings equal

- Input: s1 = "delete", s2 = "leet"
- Output: 403

Explanation: 
- Deleting "dee" from "delete" to turn the string into "let",
- adds 100[d] + 101[e] + 101[e] to the sum.
- Deleting "e" from "leet" adds 101[e] to the sum.
- At the end, both strings are equal to "let", and the answer is 100+101+101+101 = 403.
- If instead we turned both strings into "lee" or "eet", we would get answers of 433 or 417, which are higher.

### Intuition
> [!IMPORTANT]
> We need to keep the LCS, and delete other characters
>
> that mean we need to find maximum ASCII sum of LCS
>
> Final answer = total ASCII sum - maximum ASCII sum of LCS

```cpp
int minimumDeleteSum(string s1, string s2) {
    int n = s1.size(), m = s2.size();

    int dp[n+1][m+1];
    memset(dp, 0, sizeof(dp));

    unordered_set<int> st1, st2;

    for(int i=1; i<=n; i++) {
        for(int j=1; j<=m; j++) {
            if(s1[i-1] == s2[j-1]) {
                dp[i][j] = dp[i-1][j-1] + (int)s1[i-1] + (int)s2[j-1]; // important
            } else {
                dp[i][j] = max(dp[i-1][j], dp[i][j-1]);
            }
        }
    }

    int total = 0;
    for(int i=0; i<n; i++) total += s1[i];
    for(int i=0; i<m; i++) total += s2[i];

    return total - dp[n][m];
    }
```

## 5. Recap: local logging support for NIOS-XAAS October 17 | Meeting | Microsoft Teams
[Leetcode link](https://leetcode.com/problems/shortest-common-supersequence/description/)

Given two strings str1 and str2, return the shortest string that has both str1 and str2 
as subsequences. If there are multiple valid strings, return any of them.

A string s is a subsequence of string t if deleting some number of characters from t (possibly 0) 
results in the string s.

- Input: str1 = "abac", str2 = "cab"
- Output: "cabac"

### Intuition
> [!IMPORTANT]
>                         c   a   b
>                     0   0   0   0
>    i = 0        a   0   0   1   1
>    i = 1        b   0   0   1   2
>    i = 2        a   0   0   1   2
>    i = 3        c   0   1   1   2
>
>           cabac
>
> We need to backtrack on dp[][]
>
>               start from dp[n][m]
>                   if(str1[i-1] == str2[j-1])
>                       the supersequence will only have one: so pick one
>                       ans += str1[i-1]
>
>                   else if(dp[i-1][j] == dp[i][j])
>                       We need to include ans += str1[i-1], since str1[i-1] didnt match
>
>                   else
>                       ans += str2[j-1], since str2[j-1] didnt match

```cpp
string shortestCommonSupersequence(string str1, string str2) {
    
    int n = str1.size(), m = str2.size();

    int dp[n+1][m+1];
    memset(dp, 0, sizeof(dp));

    for(int i=1; i<=n; i++) {
        for(int j=1; j<=m; j++) {
            if(str1[i-1] == str2[j-1]) {
                dp[i][j] = 1 + dp[i-1][j-1];
            } else {
                dp[i][j] = max(dp[i-1][j], dp[i][j-1]);
            }
        }
    }

    string ans = "";
    int i = n, j = m;
    while(i > 0 && j > 0) {
        if(str1[i-1] == str2[j-1]) {
            ans += str1[i-1];
            i--;
            j--;
        } else if(dp[i-1][j] == dp[i][j]) {
            ans += str1[i-1];
            i--;
        } else {
            ans += str2[j-1];
            j--;
        }
    }

    while(i > 0) ans += str1[--i];
    while(j > 0) ans += str2[--j];

    reverse(ans.begin(), ans.end());
    return ans;
    }
```
